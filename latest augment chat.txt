	bool _showSetup = false;
	int _currentPage = 1; // 1 = Username, 2 = Avatar, 3 = Rules
	string _username = "";
	bool _showSteamName = true;
	bool _consentToData = true; // Default to true (full experience)
	bool _showError = false;
	string _errorMessage = "";
	string _selectedAvatarType = "steam"; // steam, upload, default
	bool _isSubmitting = false;
	bool _showLeaveWarning = false;
	PlayerSessionStats _sessionStats;

	protected override void OnStart()
	{
		Log.Info( "✅ FirstTimeSetup: Started!" );
	}

	protected override void OnUpdate()
	{
		// Check if player needs setup
		if ( _sessionStats == null )
		{
			_sessionStats = PlayerSessionStats.Local;
			if ( _sessionStats != null )
			{
				CheckIfSetupNeeded();
			}
		}
	}

	void CheckIfSetupNeeded()
	{
		// Show setup if no preferred username set
		Log.Info( $"🔍 FirstTimeSetup: Checking... PreferredUsername = '{_sessionStats.PreferredUsername}'" );

		if ( string.IsNullOrEmpty( _sessionStats.PreferredUsername ) )
		{
			_showSetup = true;
			_currentPage = 1;
			StateHasChanged();
			Log.Info( "📝 FirstTimeSetup: SHOWING MODAL!" );
		}
		else
		{
			Log.Info( $"✅ FirstTimeSetup: User already set up as '{_sessionStats.PreferredUsername}'" );
		}
	}

	string GetPageSubtitle()
	{
		return _currentPage switch
		{
			1 => "Step 1 of 3",
			2 => "Step 2 of 3",
			3 => "Step 3 of 3",
			_ => "Setup"
		};
	}

	void NextPage()
	{
		// Page 1: Validate username
		if ( _currentPage == 1 )
		{
			if ( string.IsNullOrWhiteSpace( _username ) )
			{
				_showError = true;
				_errorMessage = "⚠️ Username is required!";
				StateHasChanged();
				return;
			}

			_username = _username.Trim();

			if ( _username.Length < 3 )
			{
				_showError = true;
				_errorMessage = "⚠️ Username must be at least 3 characters!";
				StateHasChanged();
				return;
			}

			if ( !System.Text.RegularExpressions.Regex.IsMatch( _username, @"^[a-zA-Z0-9_ -]+$" ) )
			{
				_showError = true;
				_errorMessage = "⚠️ Username can only contain letters, numbers, spaces, underscores, and hyphens!";
				StateHasChanged();
				return;
			}

			if ( IsUsernameTaken( _username ) )
			{
				_showError = true;
				_errorMessage = $"⚠️ Username '{_username}' is already taken! Please choose another.";
				StateHasChanged();
				return;
			}

			_showError = false;
		}

		// Move to next page
		if ( _currentPage < 3 ) _currentPage++;
		StateHasChanged();
	}

	void PreviousPage()
	{
		if ( _currentPage > 1 ) _currentPage--;
		_showError = false;
		StateHasChanged();
	}

	void ShowLeaveWarning()
	{
		_showLeaveWarning = true;
		StateHasChanged();
	}

	void CancelLeave()
	{
		_showLeaveWarning = false;
		_consentToData = true; // Reset to accept
		StateHasChanged();
	}

	void ConfirmLeave()
	{
		Log.Info( "❌ Player declined rules and left the server" );
		// TODO: Disconnect player from server
		_showSetup = false;
		StateHasChanged();
	}

	Task CompleteSetupAsync()
	{
		if ( _isSubmitting ) return Task.CompletedTask;
		if ( !_consentToData ) return Task.CompletedTask; // Should not happen

		_isSubmitting = true;
		StateHasChanged();

		// Save data
		SavePlayerDataAsync();

		_showSetup = false;
		_isSubmitting = false;
		StateHasChanged();

		Log.Info( $"✅ Setup complete! Username: {_username}, Avatar: {_selectedAvatarType}, Consent: {_consentToData}" );

		return Task.CompletedTask;
	}

	Task SavePlayerDataAsync()
	{
		// Save to session stats
		_sessionStats.PreferredUsername = _username;

		// Save to file
		var data = new PlayerDataManager.PlayerData
		{
			SteamId = "TODO_STEAM_ID", // TODO: Get actual Steam ID
			SteamName = "TODO_STEAM_NAME",
			PreferredUsername = _username,
			PlayerName = _username,
			ShowSteamName = _showSteamName,
			AvatarType = _selectedAvatarType,
			HasCompletedSetup = true,
			ConsentToDataCollection = _consentToData,
			ConsentDate = DateTime.UtcNow,
			FirstSeen = DateTime.UtcNow,
			LastSeen = DateTime.UtcNow,
			TotalConnects = 1,
			Money = 100 // Starting money!
		};

		PlayerDataManager.SavePlayerData( data.SteamId, data );

		return Task.CompletedTask;
	}

	bool IsUsernameTaken( string username )
	{
		// Get all player data files
		try
		{
			var files = FileSystem.Data.FindFile( "PlayerData/*.json" );
			if ( files == null || !files.Any() ) return false;

			string usernameLower = username.ToLower();

			foreach ( var file in files )
			{
				try
				{
					var data = FileSystem.Data.ReadJson<PlayerDataManager.PlayerData>( $"PlayerData/{file}" );
					if ( data != null )
					{
						// Check PreferredUsername (case-insensitive)
						if ( !string.IsNullOrEmpty( data.PreferredUsername ) &&
						     data.PreferredUsername.ToLower() == usernameLower )
						{
							Log.Info( $"🚫 Username '{username}' is taken (existing: '{data.PreferredUsername}')" );
							return true;
						}

						// Also check PlayerName (case-insensitive)
						if ( !string.IsNullOrEmpty( data.PlayerName ) &&
						     data.PlayerName.ToLower() == usernameLower )
						{
							Log.Info( $"🚫 Username '{username}' is taken (existing: '{data.PlayerName}')" );
							return true;
						}
					}
				}
				catch ( Exception ex )
				{
					Log.Warning( $"⚠️ Failed to read player data file {file}: {ex.Message}" );
				}
			}

			return false;
		}
		catch ( Exception ex )
		{
			Log.Warning( $"⚠️ Failed to check username uniqueness: {ex.Message}" );
			return false; // Allow if we can't check
		}
	}



	protected override int BuildHash()
	{
		return System.HashCode.Combine( _showSetup, _username );
	}
}
